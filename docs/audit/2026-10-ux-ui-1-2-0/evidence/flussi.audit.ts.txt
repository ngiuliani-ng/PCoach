const BASE = "http://localhost:5187/PCoach/";
const A = "./#/atleta/ATL00001";
async function open(browser: import("@playwright/test").Browser, opts: BrowserContextOptions, url: string, state: MockState = richState(), noSession = false) {
  const context = await browser.newContext({ locale: "it-IT", timezoneId: "Europe/Rome", baseURL: BASE, ...opts });
  const page = await context.newPage();
  page.setDefaultTimeout(8000);
  await installMockBackend(page, state);
  if (noSession) await page.addInitScript(() => localStorage.clear());
  await page.goto(url);
  await page.locator(".main, .login-wrap").first().waitFor();
  await page.waitForTimeout(500);
  return { page, context, state };
}
const shot = (page: Page, name: string) => { fs.mkdirSync(path.join(OUT, "flussi"), { recursive: true }); return page.screenshot({ path: path.join(OUT, "flussi", name + ".png") }); };
const focusInfo = (page: Page) => page.evaluate(() => {
  const el = document.activeElement as HTMLElement | null;
  if (!el || el === document.body) return { what: "body", inView: true };
  const r = el.getBoundingClientRect();
  const name = (el.getAttribute("aria-label") || el.textContent || "").trim().replace(/\s+/g, " ").slice(0, 40);
  const header = document.querySelector(".page-header");
  const hb = header ? header.getBoundingClientRect().bottom : 0;
  return {
    what: `${el.tagName.toLowerCase()} «${name}»`,
    inView: r.right > 0 && r.left < innerWidth && r.bottom > 0 && r.top < innerHeight,
    inSidebar: !!el.closest(".sidebar"),
    underHeader: !el.closest(".page-header") && r.top < hb && r.bottom > 0 && getComputedStyle(header!).position === "sticky",
    rect: [Math.round(r.left), Math.round(r.top), Math.round(r.width), Math.round(r.height)]
  };
});

});

test("F22 segnaposto e controlli nativi nei due temi", async ({ browser }) => {
  for (const scheme of ["light", "dark"] as const) {
    const s = richState();
    const { page, context } = await open(browser, { viewport: { width: 1440, height: 900 }, colorScheme: scheme }, `${A}/profilo?sezione=sec-metodologia`, s);
    const r = await page.evaluate(() => {
      const parse = (c: string) => (c.match(/[\d.]+/g) || []).map(Number);
      const lum = (c: number[]) => { const f = (v: number) => { v /= 255; return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; }; return 0.2126 * f(c[0]) + 0.7152 * f(c[1]) + 0.0722 * f(c[2]); };
      const input = document.querySelector<HTMLInputElement>("input[placeholder^='predefinito']")!;
      const ph = getComputedStyle(input, "::placeholder").color;
      const bg = getComputedStyle(input).backgroundColor;
      const a = parse(ph), b = parse(bg);
      const alpha = a[3] ?? 1;
      const fg = [0, 1, 2].map((i) => a[i] * alpha + b[i] * (1 - alpha));
      const L1 = lum(fg), L2 = lum(b);
      return { placeholder: ph, background: bg, ratio: +((Math.max(L1, L2) + 0.05) / (Math.min(L1, L2) + 0.05)).toFixed(2), colorScheme: getComputedStyle(document.documentElement).colorScheme };
    });
    record(`F22 segnaposto ${scheme}`, r);
    await context.close();
  }
});
