async function shoot(page: Page, name: string) {
  await page.locator("#sec-feedback").screenshot({ path: `${OUT}/${name}.png` });
  const m = await page.evaluate(() => {
    const s = document.getElementById("sec-feedback")!.getBoundingClientRect();
    const draft = document.querySelector("#feedback-r-5ott-2 textarea")?.getBoundingClientRect();
    return { sectionTop: Math.round(s.top + scrollY), sectionHeight: Math.round(s.height), screens: +(s.height / innerHeight).toFixed(1), textareaTop: draft ? Math.round(draft.top + scrollY) : null, vw: innerWidth };
  });
  console.log(name, JSON.stringify(m));
}

test("interazioni", async ({ page }, info) => {
  test.skip(info.project.name !== "desktop");
  await page.setViewportSize({ width: 1440, height: 900 });
  const state = richState("giulia.ferri@example.com");
  await installMockBackend(page, state);
  // Il backend finto non gestisce «regenerate» e «create»: rispondo come il server reale.
  await page.route("**/functions/v1/weekly-feedback", async (route) => {
    const body = route.request().postDataJSON() as Record<string, unknown>;
    if (body.action === "regenerate") return route.fulfill({ status: 409, contentType: "application/json", body: JSON.stringify({ error: "La valutazione è in elaborazione: riprova tra poco." }) });
    if (body.action === "create") return route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ id: "r-28set", status: "approved" }) });
    return route.fallback();
  });
  await page.goto(`./#/atleta/${ATHLETE_ID}/panoramica`);
  const ta = page.locator("#feedback-r-5ott-2 textarea");
  await ta.fill("Testo riscritto dal coach.");
  await page.locator("#feedback-r-5ott-2").getByRole("button", { name: "Rigenera" }).click();
  await page.waitForTimeout(800);
  console.log("dopo Rigenera fallito, testo:", JSON.stringify((await ta.inputValue()).slice(0, 40)), "toast:", await page.locator(".toast").allTextContents());
  await ta.fill("Testo riscritto dal coach, seconda volta.");
  const menu = page.getByRole("button", { name: "Impostazioni" });
  await menu.click();
  await page.waitForTimeout(500);
  console.log("uscita verso Impostazioni: dialogo?", await page.getByRole("alertdialog").count(), "url", page.url());
  await page.goBack();
  await page.locator("#feedback-r-5ott-2 textarea").waitFor();
  console.log("ritorno, testo:", JSON.stringify((await page.locator("#feedback-r-5ott-2 textarea").inputValue()).slice(0, 40)));
  await page.getByRole("button", { name: "Prepara il feedback" }).click();
  await page.waitForTimeout(800);
  console.log("Prepara su settimana gia' approvata, toast:", await page.locator(".toast").allTextContents());
  const cb = page.locator("#feedback-r-5ott-2 .checkbox-line");
  console.log("casella reinvio altezza:", (await cb.boundingBox())?.height);
});
