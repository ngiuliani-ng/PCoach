async function measure(page: Page, name: string) {
  const m = await page.evaluate(() => {
    const vis = (el: Element) => {
      const r = (el as HTMLElement).getBoundingClientRect();
      const cs = getComputedStyle(el);
      return r.width > 0 && r.height > 0 && cs.visibility !== "hidden" && cs.display !== "none";
    };
    const all = (sel: string) => [...document.querySelectorAll(sel)].filter(vis);
    const controls = all("input:not([type=hidden]), select, textarea");
    const buttons = all("button, a[href], summary");
    const nameOf = (el: Element) => {
      const aria = el.getAttribute("aria-label") || (el.getAttribute("aria-labelledby") ? document.getElementById(el.getAttribute("aria-labelledby")!)?.textContent : "");
      if (aria) return aria.trim();
      const id = el.id && document.querySelector(`label[for="${el.id}"]`);
      if (id) return id.textContent!.trim();
      const wrap = el.closest("label");
      if (wrap) return wrap.textContent!.trim();
      return (el.textContent || "").trim();
    };
    const unlabeled = [...controls, ...buttons].filter((el) => !nameOf(el)).map((el) => el.outerHTML.slice(0, 120));
    const small = buttons.concat(controls).map((el) => ({ el, r: el.getBoundingClientRect() }))
      .filter(({ r }) => r.height < 24 || r.width < 24)
      .map(({ el, r }) => `${el.tagName.toLowerCase()} "${nameOf(el).slice(0, 40)}" ${Math.round(r.width)}x${Math.round(r.height)}`);
    const under44 = buttons.concat(controls).filter((el) => el.getBoundingClientRect().height < 44).length;
    const headings = [...document.querySelectorAll("h1,h2,h3,h4")].filter(vis).map((h) => `${h.tagName}:${h.textContent!.trim().slice(0, 50)}`);
    const helpers = all(".helper-text, .field-help, .anomaly-detail");
    const helperChars = helpers.reduce((n, el) => n + (el.textContent || "").trim().length, 0);
    const smallText = [...document.querySelectorAll("body *")].filter((el) => vis(el) && el.childNodes.length && [...el.childNodes].some((c) => c.nodeType === 3 && c.textContent!.trim()) && parseFloat(getComputedStyle(el).fontSize) < 12).length;
    return {
      viewport: `${innerWidth}x${innerHeight}`,
      docHeight: document.documentElement.scrollHeight,
      screens: +(document.documentElement.scrollHeight / innerHeight).toFixed(1),
      hOverflow: document.documentElement.scrollWidth > document.documentElement.clientWidth,
      controls: controls.length, buttons: buttons.length,
      details: document.querySelectorAll("details").length, detailsOpen: document.querySelectorAll("details[open]").length,
      helpers: helpers.length, helperChars,
      under44, small, unlabeled, headings, smallText
    };
  });
  fs.appendFileSync(`${OUT}/measures.jsonl`, JSON.stringify({ name, ...m }) + "\n");
  return m;
}

async function shot(page: Page, name: string, full = true) {
  await page.waitForTimeout(400);
  await page.screenshot({ path: `${OUT}/${name}.png`, fullPage: full });
  await measure(page, name);
}

async function openSidebarIfMobile(page: Page) {
  const menu = page.getByRole("button", { name: "Apri la barra laterale" });
  if (await menu.isVisible().catch(() => false)) await menu.click();
}

  { tag: "1440", width: 1440, height: 900, touch: false },
  { tag: "1024", width: 1024, height: 768, touch: true },
  { tag: "768", width: 768, height: 1024, touch: true },

  { tag: "320", width: 320, height: 640, touch: true }

  test.describe(`vp ${vp.tag}`, () => {

    test(`pagine ${vp.tag}`, async ({ page }) => {
      const errors: string[] = [];
      page.on("console", (m) => { if (m.type() === "error") errors.push(m.text()); });
      page.on("pageerror", (e) => errors.push(String(e)));

      await installMockBackend(page, state);

      await page.waitForTimeout(800);
      await shot(page, `${v}-home`);
      if (vp.width < 720) {
        await openSidebarIfMobile(page);

        await page.keyboard.press("Escape");
      }

      await page.goto(`./#/atleta/${ATHLETE_ID}/panoramica`);
      await page.waitForTimeout(1200);
      await shot(page, `${v}-panoramica`);

      await page.goto(`./#/atleta/${ATHLETE_ID}/piano`);
      await page.waitForTimeout(800);
      await shot(page, `${v}-piano-settimana`);
      await page.getByRole("button", { name: "Storico" }).click();

      await page.waitForTimeout(600);

test.describe("tastiera desktop", () => {
  test.use({ viewport: { width: 1440, height: 900 } });
  test("ordine del focus nella scheda", async ({ page }) => {
    await installMockBackend(page, richState());
    await page.goto(`./#/atleta/${ATHLETE_ID}/panoramica`);
    await page.waitForTimeout(1000);
    const seq: string[] = [];
    for (let i = 0; i < 25; i++) {
      await page.keyboard.press("Tab");
      seq.push(await page.evaluate(() => {
        const el = document.activeElement as HTMLElement;
        const cs = getComputedStyle(el);
        const ring = cs.outlineStyle !== "none" && cs.outlineWidth !== "0px" ? "outline" : cs.boxShadow !== "none" ? "shadow" : "NESSUNO";
        return `${el.tagName.toLowerCase()}[${(el.getAttribute("aria-label") || el.textContent || "").trim().slice(0, 40)}] focus:${ring}`;
      }));
    }
    fs.appendFileSync(`${OUT}/measures.jsonl`, JSON.stringify({ name: "focus-order-panoramica", seq }) + "\n");
  });

  test("zoom 200% simulato (640x450)", async ({ page }) => {
    await page.setViewportSize({ width: 640, height: 450 });
    await installMockBackend(page, richState());
    await page.goto(`./#/atleta/${ATHLETE_ID}/piano`);
    await page.waitForTimeout(1000);
    await shot(page, `zoom200-piano`, false);
    await page.goto(`./#/impostazioni`);
    await page.waitForTimeout(800);
    await shot(page, `zoom200-impostazioni`, false);
  });
});
